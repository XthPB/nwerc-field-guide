Functional graph distances and cycles

Reuse the tree and cycle decomposition. For a cycle DP enforce consistency between the first and last decisions on the cycle.

From the ICPC Code Templates cheat sheet.

// Functional graph: every node v has exactly ONE out-edge nxt[v]
//   (permutations, "teleporters", f(x) iterated, linked lists).
// USE: which nodes lie on cycles; where you are after k steps
//   (k up to 1e18) -> binary lifting. TIME: cycles O(n).
// onCycle[v] = true if v lies on a cycle
vector<bool> findCycleNodes(const vector<int> &nxt) {
    int n = nxt.size();
    vector<int> state(n, 0);   // 0 new, 1 on current walk, 2 done
    vector<bool> onCycle(n, false);
    for (int s = 0; s < n; s++) {
        vector<int> path;
        int v = s;
        while (state[v] == 0) {
            state[v] = 1;
            path.push_back(v);
            v = nxt[v];
        }
        if (state[v] == 1)     // reached our own walk: new cycle at v
            for (int i = path.size() - 1; i >= 0; i--) {
                onCycle[path[i]] = true;
                if (path[i] == v) break;
            }
        for (int u : path) state[u] = 2;
    }
    return onCycle;
}
// node after k steps (k < 2^LOG): O(n LOG) build, O(LOG) query
const int LOG = 60;            // 30 is enough for k <= 1e9
vector<vector<int>> buildJumps(const vector<int> &nxt) {
    vector<vector<int>> jump(LOG, nxt);  // 2^j steps from v
    for (int j = 1; j < LOG; j++)
        for (int v = 0; v < (int)nxt.size(); v++)
            jump[j][v] = jump[j - 1][jump[j - 1][v]];
    return jump;
}
int kthNext(const vector<vector<int>> &jump, int v, ll k) {
    for (int j = 0; j < LOG; j++)
        if ((k >> j) & 1) v = jump[j][v];
    return v;
}
// fewest steps from x to reach a node >= y when nxt[v] >= v (steps
//   only go right, nxt[v] == v means stuck). -1 if never. O(LOG).
ll fewestSteps(const vector<vector<int>> &jump, int x, int y) {
    if (x >= y) return 0;
    ll c = 0;
    for (int j = LOG - 1; j >= 0; j--)
        if (jump[j][x] < y) { x = jump[j][x]; c += 1LL << j; }
    return jump[0][x] >= y ? c + 1 : -1;
}
