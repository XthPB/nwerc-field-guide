Rollback XOR potentials for a dynamic forest

find(v) returns the root and the XOR from v to that root. Adding an edge inside a component returns the fundamental cycle XOR, including zero. A joining edge returns no value. Pair each DFS checkpoint with a separate checkpoint for the linear basis. Use one global basis only when the query graph is connected.

struct UndoXorForest {
    using Word = unsigned long long;
    struct Change { int child, parent, oldSize; };
    std::vector<int> up, size;
    std::vector<Word> diff;
    std::vector<Change> log;
    explicit UndoXorForest(int n) : up(n), size(n,1), diff(n,0) {
        std::iota(up.begin(),up.end(),0);
    }
    std::pair<int,Word> find(int u) const {
        Word value = 0;
        while (u != up[u]) {
            value ^= diff[u];
            u = up[u];
        }
        return {u,value};
    }
    int checkpoint() const { return (int)log.size(); }
    std::optional<Word> add(int u, int v, Word w) {
        auto [a, x] = find(u);
        auto [b, y] = find(v);
        Word link = x ^ w ^ y;
        if (a == b) return link;
        if (size[a] < size[b]) std::swap(a,b);
        log.push_back({b,a,size[a]});
        up[b] = a; diff[b] = link; size[a] += size[b];
        return std::nullopt;
    }
    void rollback(int mark) {
        while ((int)log.size() > mark) {
            auto [b,a,old] = log.back();
            log.pop_back();
            up[b] = b; diff[b] = 0; size[a] = old;
        }
    }
};
