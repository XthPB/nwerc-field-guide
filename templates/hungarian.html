Hungarian assignment

Use a rectangular or padded assignment model with the correct dimensions and feasibility assumptions.

From the ICPC Code Templates section “Hungarian Algorithm (assignment)”. Use the code with the types and conventions shown in the excerpt.

The listing includes its required numeric constant. Choose bounds and tolerances for the input scale.

constexpr long long INF = (1LL << 62);
// Hungarian algorithm: assignment problem. n rows (workers) to m
//   columns (jobs), n <= m, every row gets a different column,
//   minimise the total cost. Maximise: negate the costs.
// NEEDS: cost matrix a[n][m] (0-based, any sign), INF (template).
// TIME: O(n^2 m) (n = m = 500: ~0.1 s). Faster than min-cost flow.
// returns {min cost, col[i] = column given to row i}
pair<ll, vector<int>> hungarian(const vector<vector<ll>> &a) {
    int n = a.size(), m = a[0].size();
    vector<ll> u(n + 1), v(m + 1);           // potentials
    vector<int> p(m + 1), way(m + 1);        // p[j] = row of column j
    for (int i = 1; i <= n; i++) {
        p[0] = i;
        int j0 = 0;
        vector<ll> minv(m + 1, INF);
        vector<bool> used(m + 1, false);
        do {
            used[j0] = true;
            int i0 = p[j0], j1 = 0;
            ll delta = INF;
            for (int j = 1; j <= m; j++)
                if (!used[j]) {
                    ll cur = a[i0 - 1][j - 1] - u[i0] - v[j];
                    if (cur < minv[j]) { minv[j] = cur; way[j] = j0; }
                    if (minv[j] < delta) { delta = minv[j]; j1 = j; }
                }
            for (int j = 0; j <= m; j++)
                if (used[j]) { u[p[j]] += delta; v[j] -= delta; }
                else minv[j] -= delta;
            j0 = j1;
        } while (p[j0] != 0);
        do {
            int j1 = way[j0];
            p[j0] = p[j1];
            j0 = j1;
        } while (j0);
    }
    vector<int> col(n);
    for (int j = 1; j <= m; j++)
        if (p[j]) col[p[j] - 1] = j - 1;
    return {-v[0], col};
}
