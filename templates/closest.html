Closest and farthest pairs

// Farthest pair (diameter) of a point set: rotating calipers over
//   the convex hull above. TIME: O(n log n). Returns squared dist.
ll farthestPair(const vector<Point> &pts) {
    vector<Point> h = convexHull(pts);
    int k = h.size();
    if (k == 1) return 0;
    if (k == 2) return dist2(h[0], h[1]);
    ll best = 0;
    for (int i = 0, j = 1; i < k; i++) {
        Point e = h[(i + 1) % k] - h[i];          // edge i -> i+1
        while (cross(e, h[(j + 1) % k] - h[j]) > 0) j = (j + 1) % k;
        best = max({best, dist2(h[i], h[j]),
                    dist2(h[(i + 1) % k], h[j])});
    }
    return best;
}
// Closest pair: sweep by x, keep points with x within d in a set
//   ordered by y. NEEDS: >= 2 points. TIME: O(n log n). Squared.
ll closestPair(vector<Point> pts) {
    sort(pts.begin(), pts.end());
    set<pair<ll, ll>> active;                 // {y, x}
    ll best = LLONG_MAX;
    int left = 0;
    for (Point p : pts) {
        ll d = sqrtl((long double)best) + 1;   // current distance
        while (p.x - pts[left].x > d) {
            active.erase({pts[left].y, pts[left].x});
            left++;
        }
        auto it = active.lower_bound({p.y - d, LLONG_MIN});
        for (; it != active.end() && it->first <= p.y + d; it++) {
            ll dx = p.x - it->second, dy = p.y - it->first;
            best = min(best, dx * dx + dy * dy);
        }
        active.insert({p.y, p.x});
    }
    return best;
}
// ALL pairs within distance R: grid cells of side R (floorDiv), map
//   cell -> points, check the 3x3 cells around each point; fast when
//   each cell holds O(1) points. 3D closest pair: same sweep + dz.
