Rectangle union by sweep

Compress endpoints and use the original coordinate gaps for lengths.

From the ICPC Code Templates section “Union of Rectangles (area sweep)”. Use the code with the types and conventions shown in the excerpt.

// Area of the UNION of axis-parallel rectangles: sweep a vertical
//   line over x; a segment tree over the sorted y values tracks how
//   much of the line is covered. TIME: O(n log n).
// NEEDS: {x1, y1, x2, y2} with x1 < x2, y1 < y2; area < 9e18.
struct CoverTree {
    vector<ll> ys;                     // sorted distinct y values
    vector<int> cnt;                   // rectangles covering node
    vector<ll> len;                    // covered length in node
    CoverTree(const vector<ll> &y)
        : ys(y), cnt(4 * y.size(), 0), len(4 * y.size(), 0) {}
    // add v to the strips ys[ql]..ys[qr]; node covers ys[l]..ys[r]
    void update(int node, int l, int r, int ql, int qr, int v) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) cnt[node] += v;
        else {
            int m = (l + r) / 2;
            update(2 * node, l, m, ql, qr, v);
            update(2 * node + 1, m, r, ql, qr, v);
        }
        if (cnt[node] > 0) len[node] = ys[r] - ys[l];
        else if (r - l == 1) len[node] = 0;
        else len[node] = len[2 * node] + len[2 * node + 1];
    }
};
ll unionArea(const vector<array<ll, 4>> &rects) {
    if (rects.empty()) return 0;
    vector<ll> ys;
    vector<array<ll, 4>> events;       // {x, +1 or -1, y1, y2}
    for (auto [x1, y1, x2, y2] : rects) {
        ys.push_back(y1); ys.push_back(y2);
        events.push_back({x1, 1, y1, y2}); events.push_back({x2, -1, y1, y2});
    }
    sort(ys.begin(), ys.end());
    ys.erase(unique(ys.begin(), ys.end()), ys.end());
    sort(events.begin(), events.end());
    CoverTree t(ys);
    auto idx = [&](ll y) {
        return int(lower_bound(ys.begin(), ys.end(), y) - ys.begin());
    };
    ll area = 0;
    for (int i = 0; i < (int)events.size(); i++) {
        if (i > 0) area += t.len[1] * (events[i][0] - events[i - 1][0]);
        t.update(1, 0, ys.size() - 1, idx(events[i][2]),
                 idx(events[i][3]), events[i][1]);
    }
    return area;
}
