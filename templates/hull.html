Convex hull and angular order

// Convex hull (monotone chain). USE: smallest convex polygon around
//   points, farthest pair (on hull), perimeter of a fence.
// TIME: O(n log n). Counter-clockwise, no collinear points on it.
vector<Point> convexHull(vector<Point> pts) {
    sort(pts.begin(), pts.end());
    pts.erase(unique(pts.begin(), pts.end()), pts.end());
    if (pts.size() < 3) return pts;
    vector<Point> hull;
    for (int pass = 0; pass < 2; pass++) {   // lower, then upper
        int start = hull.size();
        for (Point p : pts) {
            while ((int)hull.size() >= start + 2 &&
                   cross(hull[hull.size() - 2], hull.back(), p) <= 0)
                hull.pop_back();
            hull.push_back(p);
        }
        hull.pop_back();       // it starts the other half
        reverse(pts.begin(), pts.end());
    }
    return hull;
}
// Point in polygon: ANY simple polygon (convex or not), vertices
//   in order (either direction). TIME: O(n), exact for integers.
// returns 1 inside, 0 on the boundary, -1 outside
int pointInPolygon(Point p, const vector<Point> &poly) {
    int n = poly.size();
    bool inside = false;
    for (int i = 0; i < n; i++) {
        Point a = poly[i], b = poly[(i + 1) % n];
        if (onSegment(p, a, b)) return 0;
        // edge crosses the horizontal line through p ...
        if ((a.y > p.y) != (b.y > p.y)) {
            // ... to the right of p: flip
            if ((cross(a, b, p) > 0) == (b.y > a.y)) inside = !inside;
        }
    }
    return inside ? 1 : -1;
}
// sort vectors by angle in [0, 2pi) from the +x axis, exact,
// (0, 0) not allowed: sort(v.begin(), v.end(), angleLess);
bool upperHalf(Point p) { return p.y > 0 || (p.y == 0 && p.x > 0); }
bool angleLess(Point a, Point b) {
    if (upperHalf(a) != upperHalf(b)) return upperHalf(a);
    return cross(a, b) > 0;
}
