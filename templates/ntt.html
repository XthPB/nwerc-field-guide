Number-theoretic transform

Use this for ordinary polynomial convolution with the supported modulus and transform length. Choose an appropriate root and power-of-two length and bring modular exponentiation from the linked arithmetic section.

From the ICPC Code Templates cheat sheet.

// NTT: multiply two polynomials mod 998244353 in O(n log n).
// USE: convolution c[k] = sum of a[i] * b[k - i]: #ways to get each
//   sum from two lists, products of generating functions.
// NEEDS: power() from Modular Arithmetic; coefficients in [0, MOD);
//   result length < 2^23. Exact answers that may exceed the mod:
//   repeat with other NTT primes (e.g. 167772161) and use CRT.
const ll NTT_MOD = 998244353;      // = 119 * 2^23 + 1, root 3
void ntt(vector<ll> &a, bool invert) {
    int n = a.size();
    for (int i = 1, j = 0; i < n; i++) {   // bit-reversal order
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        ll w = power(3, (NTT_MOD - 1) / len, NTT_MOD);
        if (invert) w = power(w, NTT_MOD - 2, NTT_MOD);
        for (int i = 0; i < n; i += len) {
            ll wn = 1;
            for (int j = 0; j < len / 2; j++) {
                ll u = a[i + j], v = a[i + j + len / 2] * wn % NTT_MOD;
                a[i + j] = (u + v) % NTT_MOD;
                a[i + j + len / 2] = (u - v + NTT_MOD) % NTT_MOD;
                wn = wn * w % NTT_MOD;
            }
        }
    }
    if (invert) {
        ll inv = power(n, NTT_MOD - 2, NTT_MOD);
        for (ll &x : a) x = x * inv % NTT_MOD;
    }
}
vector<ll> polyMultiply(vector<ll> a, vector<ll> b) {
    if (a.empty() || b.empty()) return {};
    int need = a.size() + b.size() - 1, n = 1;
    while (n < need) n <<= 1;
    a.resize(n);
    b.resize(n);
    ntt(a, false);
    ntt(b, false);
    for (int i = 0; i < n; i++) a[i] = a[i] * b[i] % NTT_MOD;
    ntt(a, true);
    a.resize(need);
    return a;
}
