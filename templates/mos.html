Mo’s offline queries

Supply add, remove and get_answer for the statistic in your problem. The example supplies the query ordering and pointer movement.

// Mo's algorithm: answer OFFLINE range queries [l, r] when you can
//   add/remove one element quickly (distinct count, mode, pairs...).
// NEEDS: all queries known in advance, no updates, 0-based inclusive.
// TIME: O((n + q) * sqrt(n)) add/remove calls.
// SET block_size = max(1, (int)sqrt(n)) BEFORE sorting.
void add(int idx);      // TODO: add a[idx] to the data structure
void remove(int idx);   // TODO: remove a[idx]
int get_answer();       // TODO: answer for the current range
int block_size;
struct Query {
    int l, r, idx;
    bool operator<(Query other) const
    {
        return make_pair(l / block_size, r) <
               make_pair(other.l / block_size, other.r);
    }
};
vector<int> mo_s_algorithm(vector<Query> queries) {
    vector<int> answers(queries.size());
    sort(queries.begin(), queries.end());
    // TODO: initialize data structure
    int cur_l = 0;
    int cur_r = -1;
    // invariant: data structure reflects the range [cur_l, cur_r]
    for (Query q : queries) {
        while (cur_l > q.l) {
            cur_l--;
            add(cur_l);
        }
        while (cur_r < q.r) {
            cur_r++;
            add(cur_r);
        }
        while (cur_l < q.l) {
            remove(cur_l);
            cur_l++;
        }
        while (cur_r > q.r) {
            remove(cur_r);
            cur_r--;
        }
        answers[q.idx] = get_answer();
    }
    return answers;
}
// SQRT SPLIT BY SIZE (not tree HLD): S = total size of all sets or
//   adjacency lists, B ~ sqrt(S). HEAVY (size >= B): at most S/B of
//   them, handle each with one O(S) scan. LIGHT: brute force all pairs
//   inside, sum of |s|^2 <= B * S. Total O(S sqrt S).
// e.g. two sets sharing 2 values; triangles: orient each edge to the
//   higher (degree, id) end, out-degree <= sqrt(2m), O(m sqrt m).
