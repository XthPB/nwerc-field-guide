Rollback disjoint-set union

Save a checkpoint, unite vertices, then restore precisely that checkpoint. Union by size bounds find depth while each successful union records one reversible change.

Guide companion to the cheat sheet’s union-find routine. Include vector, utility, numeric and algorithm. Checkpoints belong to the current history.

struct RollbackDSU {
    std::vector<int> parent, size;
    std::vector<std::pair<int,int>> history;
    explicit RollbackDSU(int n) : parent(n), size(n, 1) {
        std::iota(parent.begin(), parent.end(), 0);
    }
    int find(int x) const {
        while (x != parent[x]) x = parent[x];
        return x;
    }
    int snapshot() const { return (int)history.size(); }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (size[a] < size[b]) std::swap(a, b);
        history.emplace_back(b, size[a]);
        parent[b] = a;
        size[a] += size[b];
        return true;
    }
    void rollback(int checkpoint) {
        while ((int)history.size() > checkpoint) {
            auto [b, oldSize] = history.back();
            history.pop_back();
            size[parent[b]] = oldSize;
            parent[b] = b;
        }
    }
};
