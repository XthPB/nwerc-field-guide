Circle intersections

Classify containment, tangency, coincident centers and disjointness before solving for intersection points.

From the ICPC Code Templates section “Geometry: Circle Intersections”. Use the code with the types and conventions shown in the excerpt.

The listing includes its required numeric constant. Choose bounds and tolerances for the input scale.

constexpr long double EPS = 1e-9L;
// Circle intersections (long double). USE: where circles meet,
//   line/segment hitting a circle. TIME: O(1). Tangent -> 1 point.
// NEEDS: EPS = 1e-9 from the Trig section.
typedef long double ld;
struct PointD { ld x, y; };
// circle (center c, radius r) and line through a, b: 0, 1, 2 points
// each point = a + t (b - a); 0 <= t <= 1 means on segment [a, b]
vector<PointD> circleLine(PointD c, ld r, PointD a, PointD b) {
    ld dx = b.x - a.x, dy = b.y - a.y;
    ld fx = a.x - c.x, fy = a.y - c.y;
    ld A = dx * dx + dy * dy;
    ld B = 2 * (fx * dx + fy * dy);
    ld C = fx * fx + fy * fy - r * r;
    ld disc = B * B - 4 * A * C;
    if (disc < -EPS) return {};
    disc = sqrtl(max((ld)0, disc));
    ld t1 = (-B - disc) / (2 * A), t2 = (-B + disc) / (2 * A);
    PointD p1 = {a.x + t1 * dx, a.y + t1 * dy};
    PointD p2 = {a.x + t2 * dx, a.y + t2 * dy};
    if (disc < EPS) return {p1};
    return {p1, p2};
}
// two circles: 0, 1 or 2 points ({} also for identical circles)
vector<PointD> circleCircle(PointD c1, ld r1, PointD c2, ld r2) {
    ld dx = c2.x - c1.x, dy = c2.y - c1.y;
    ld d = sqrtl(dx * dx + dy * dy);
    if (d < EPS || d > r1 + r2 + EPS || d < fabsl(r1 - r2) - EPS)
        return {};
    ld a = (r1 * r1 - r2 * r2 + d * d) / (2 * d);  // c1 to chord
    ld h = sqrtl(max((ld)0, r1 * r1 - a * a));     // half chord
    ld mx = c1.x + a * dx / d, my = c1.y + a * dy / d;
    if (h < EPS) return {{mx, my}};
    return {{mx - h * dy / d, my + h * dx / d},
            {mx + h * dy / d, my - h * dx / d}};
}
