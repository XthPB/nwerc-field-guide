Bottom-up segment tree

The original combine is addition over a half-open interval. For a min, max, witness or matrix payload, update the stored type, combine and empty identity together. Preserve combination order for matrices.

From the ICPC Code Templates cheat sheet.

// Segment tree with 2n nodes (bottom-up, no recursion).
// USE: point update + range query for an ASSOCIATIVE operation:
//   sum, min, max, gcd, xor, and, or (also non-commutative ones).
// NEEDS: 0-based; query(l, r) is HALF-OPEN: a[l] .. a[r-1].
// TIME: build O(n), update and query O(log n).  MEMORY: 2n.
// Change combine() and ID together. ID (identity): sum/xor/or/gcd 0,
//   min LLONG_MAX, max LLONG_MIN, and -1 (all bits), product 1.
struct SegTree {
    const ll ID = 0;              // declared before t (init order)
    ll combine(ll a, ll b) { return a + b; }
    int n;
    vector<ll> t;                       // leaves t[n..2n-1], root t[1]
    SegTree(const vector<ll> &a) : n(a.size()), t(2 * a.size(), ID) {
        for (int i = 0; i < n; i++) t[n + i] = a[i];
        for (int i = n - 1; i > 0; i--)
            t[i] = combine(t[2 * i], t[2 * i + 1]);
    }
    void update(int pos, ll value) {    // a[pos] = value
        pos += n;
        t[pos] = value;
        for (pos /= 2; pos >= 1; pos /= 2)
            t[pos] = combine(t[2 * pos], t[2 * pos + 1]);
    }
    ll query(int l, int r) {            // combine of a[l..r-1]
        ll left = ID, right = ID;
        for (l += n, r += n; l < r; l /= 2, r /= 2) {
            if (l & 1) left = combine(left, t[l++]);
            if (r & 1) right = combine(t[--r], right);
        }
        return combine(left, right);
    }
};
// add to a[pos] instead of set: update(pos, query(pos, pos+1) + v)
