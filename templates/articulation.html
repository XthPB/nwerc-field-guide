Articulation points

The DFS root has its own child-count condition.

From the ICPC Code Templates section “Articulation Points”. Use the code with the types and conventions shown in the excerpt.

// Articulation points: vertices whose removal disconnects the graph.
// NEEDS: UNDIRECTED, nodes 0..n-1. TIME: O(V + E), recursive.
// Globals clash with Bridges, SCC and LCA: one per program.
// isCut[v] is set once even if found several times.
vector<bool> isCut;
void IS_CUTPOINT(int v) { isCut[v] = true; }
int n; // number of nodes
vector<vector<int>> adj; // adjacency list of graph
vector<bool> visited;
vector<int> tin, low;
int timer;
void dfs(int v, int p = -1) {
    visited[v] = true;
    tin[v] = low[v] = timer++;
    int children=0;
    for (int to : adj[v]) {
        if (to == p) continue;
        if (visited[to]) {
            low[v] = min(low[v], tin[to]);
        } else {
            dfs(to, v);
            low[v] = min(low[v], low[to]);
            if (low[to] >= tin[v] && p!=-1)
                IS_CUTPOINT(v);
            ++children;
        }
    }
    if(p == -1 && children > 1)
        IS_CUTPOINT(v);
}
void find_cutpoints() {
    timer = 0;
    isCut.assign(n, false);
    visited.assign(n, false);
    tin.assign(n, -1);
    low.assign(n, -1);
    for (int i = 0; i < n; ++i) {
        if (!visited[i])
            dfs (i);
    }
}
