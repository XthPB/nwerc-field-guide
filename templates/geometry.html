Exact integer geometry

Use dot and cross products for structural tests. For route costs use the distance definition required by the problem and a sufficiently wide type. Exact squared distance and Euclidean length may give different optima.

From the ICPC Code Templates cheat sheet.

// Integer geometry: exact, no EPS. |coord| <= 1e9 keeps cross in ll.
// cross(a, b) = |a||b| sin(angle), dot(a, b) = |a||b| cos(angle)
struct Point {
    ll x, y;
    Point operator+(Point o) const { return {x + o.x, y + o.y}; }
    Point operator-(Point o) const { return {x - o.x, y - o.y}; }
    bool operator<(Point o) const {
        return x < o.x || (x == o.x && y < o.y);
    }
    bool operator==(Point o) const { return x == o.x && y == o.y; }
};
ll dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
ll cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
// > 0: o -> a -> b turns left (counter-clockwise), < 0 right,
// 0 collinear
ll cross(Point o, Point a, Point b) { return cross(a - o, b - o); }
ll dist2(Point a, Point b) { return dot(a - b, a - b); }  // squared
int sgn(ll v) { return (v > 0) - (v < 0); }

These are contest snippets. Bring the declarations they use into your file, including standard headers, the ll alias or the modulus where applicable.
