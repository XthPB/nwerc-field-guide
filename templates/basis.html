Linear XOR basis

Basis insertion tests independence. For a compressed XOR DP, also encode allowed values in coordinates of the independent basis. Rank controls the transformed table size.

From the ICPC Code Templates cheat sheet.

// XOR basis (linear algebra over bits). USE: max xor of a SUBSET,
//   "can x be made as xor of some elements", #distinct subset xors.
//   (max xor of a PAIR: use the XOR trie instead.)
// NEEDS: 0 <= values < 2^61. TIME: O(61) per insert/query.
struct XorBasis {
    ll basis[61] = {};     // basis[b] = vector with highest bit b
    int size = 0;          // #different subset xors = 2^size
    bool insert(ll x) {    // false if x was already reachable
        for (int b = 60; b >= 0; b--) {
            if (!((x >> b) & 1)) continue;
            if (basis[b] == 0) { basis[b] = x; size++; return true; }
            x ^= basis[b];
        }
        return false;
    }
    bool canMake(ll x) {
        for (int b = 60; b >= 0; b--)
            if ((x >> b) & 1) {
                if (basis[b] == 0) return false;
                x ^= basis[b];
            }
        return true;
    }
    ll maxXor() {
        ll r = 0;
        for (int b = 60; b >= 0; b--) r = max(r, r ^ basis[b]);
        return r;
    }
};
