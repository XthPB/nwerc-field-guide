Exact integer arithmetic

Floor and ceiling division differ from truncation for negative values.

From the ICPC Code Templates section “Integer Helpers (floor/ceil division, exact sqrt, int128)”. Use the code with the types and conventions shown in the excerpt.

// Integer helpers. C++ "/" rounds toward 0; these round down / up
//   also for negative numbers. isqrt is exact (sqrt() can be off).
ll floorDiv(ll a, ll b) {
    return a / b - ((a % b != 0) && ((a < 0) != (b < 0)));
}
ll ceilDiv(ll a, ll b) {
    return a / b + ((a % b != 0) && ((a < 0) == (b < 0)));
}
ll isqrt(ll n) {                   // exact floor(sqrt(n)), n >= 0
    ll r = sqrtl(n);
    while ((__int128)r * r > n) r--;
    while ((__int128)(r + 1) * (r + 1) <= n) r++;
    return r;
}
void print128(__int128 x) {        // cout cannot print __int128
    if (x < 0) { cout << '-'; x = -x; }
    if (x > 9) print128(x / 10);
    cout << char('0' + x % 10);
}
