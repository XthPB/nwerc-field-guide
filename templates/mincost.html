Minimum-cost flow

This cheat-sheet implementation uses shortest augmenting paths with SPFA and reverse edges. It assumes there is no relevant negative-cost cycle. The lesson also explains the alternative with Dijkstra and feasible potentials.

From the ICPC Code Templates section “Min-Cost Max-Flow”. Use the code with the types and conventions shown in the excerpt.

// Min-cost max-flow. USE: assignment / matching with costs,
//   "send k units as cheaply as possible", transport problems.
// NEEDS: directed edges (cap, cost), nodes 0..n-1. Negative costs
//   OK if there is no negative cycle.
// TIME: O(F * V * E) worst case (F = flow); fine for V ~ 500,
//   E ~ 5000. solve(s, t) -> {flow, cost}; maxFlow stops early.
struct MinCostFlow {
    struct Edge { int to; ll cap, cost; };
    vector<Edge> edges;            // edge i and i ^ 1 are a pair
    vector<vector<int>> adj;
    MinCostFlow(int n) : adj(n) {}
    void addEdge(int u, int v, ll cap, ll cost) {
        adj[u].push_back(edges.size());
        edges.push_back({v, cap, cost});
        adj[v].push_back(edges.size());
        edges.push_back({u, 0, -cost});
    }
    pair<ll, ll> solve(int s, int t, ll maxFlow = LLONG_MAX) {
        int n = adj.size();
        ll flow = 0, cost = 0;
        while (flow < maxFlow) {
            vector<ll> dist(n, LLONG_MAX);
            vector<int> viaEdge(n, -1);
            vector<bool> inQueue(n, false);
            queue<int> q;
            dist[s] = 0;
            q.push(s);
            while (!q.empty()) {       // cheapest path in residual
                int u = q.front();
                q.pop();
                inQueue[u] = false;
                for (int id : adj[u]) {
                    Edge &e = edges[id];
                    if (e.cap > 0 && dist[u] + e.cost < dist[e.to]) {
                        dist[e.to] = dist[u] + e.cost;
                        viaEdge[e.to] = id;
                        if (!inQueue[e.to]) {
                            inQueue[e.to] = true;
                            q.push(e.to);
                        }
                    }
                }
            }
            if (dist[t] == LLONG_MAX) break;
            ll push = maxFlow - flow;
            for (int v = t; v != s; v = edges[viaEdge[v] ^ 1].to)
                push = min(push, edges[viaEdge[v]].cap);
            for (int v = t; v != s; v = edges[viaEdge[v] ^ 1].to) {
                edges[viaEdge[v]].cap -= push;
                edges[viaEdge[v] ^ 1].cap += push;
            }
            flow += push;
            cost += push * dist[t];
        }
        return {flow, cost};
    }
};
// assignment: s->worker (1, 0), worker->job (1, c), job->t (1, 0)
// maximum cost/profit: use -c and negate the answer; if the flow
//   amount is free, add "if (dist[t] >= 0) break;" after the SPFA
