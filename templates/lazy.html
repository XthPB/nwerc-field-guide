Lazy range updates

The sheet implements range addition and sum. For range-add and range-max store the maximum as the node aggregate and add the tag once to it, while preserving lazy propagation.

From the ICPC Code Templates cheat sheet.

// Lazy segment tree: RANGE ADD + RANGE SUM (recursive, 4n nodes).
// USE: many "add v to a[l..r]" mixed with "sum of a[l..r]".
// NEEDS: 0-based, INCLUSIVE [l, r], non-empty array.
// TIME: O(log n) per update/query.  RANGE MIN instead of sum:
//   "combine" lines: x + y -> min(x, y); "size" line: drop
//   "(end - start + 1) *"; "identity" line: return LLONG_MAX.
class SegmentTree {
private:
    vector<ll> tree, lazy;
    int n;
    void buildTree(const vector<ll>& arr, int start, int end,
                   int node) {
        if (start == end) {
            tree[node] = arr[start];
            return;
        }
        int mid = (start + end) / 2;
        buildTree(arr, start, mid, 2 * node + 1);
        buildTree(arr, mid + 1, end, 2 * node + 2);
        tree[node] = tree[2 * node + 1] + tree[2 * node + 2]; // combine
    }
    void propagate(int start, int end, int node) {
        if (lazy[node] != 0) {
            tree[node] += (end - start + 1) * lazy[node];      // size
            if (start != end) {
                lazy[2 * node + 1] += lazy[node];
                lazy[2 * node + 2] += lazy[node];
            }
            lazy[node] = 0;
        }
    }
    void updateRange(int l, int r, ll val, int start, int end,
                     int node) {
        propagate(start, end, node);
        if (start > end || start > r || end < l) return;
        if (start >= l && end <= r) {
            lazy[node] += val;
            propagate(start, end, node);
            return;
        }
        int mid = (start + end) / 2;
        updateRange(l, r, val, start, mid, 2 * node + 1);
        updateRange(l, r, val, mid + 1, end, 2 * node + 2);
        tree[node] = tree[2 * node + 1] + tree[2 * node + 2]; // combine
    }
    ll queryRange(int l, int r, int start, int end, int node) {
        propagate(start, end, node);
        if (start > end || start > r || end < l) return 0;  // identity
        if (start >= l && end <= r) return tree[node];
        int mid = (start + end) / 2;
        ll leftQuery = queryRange(l, r, start, mid, 2 * node + 1);
        ll rightQuery = queryRange(l, r, mid + 1, end, 2 * node + 2);
        return leftQuery + rightQuery;                       // combine
    }
public:
    SegmentTree(const vector<ll>& arr) {
        n = arr.size();
        tree.resize(4 * n, 0);
        lazy.resize(4 * n, 0);
        buildTree(arr, 0, n - 1, 0);
    }
    void update(int l, int r, ll val) {     // a[l..r] += val
        updateRange(l, r, val, 0, n - 1, 0);
    }
    ll query(int l, int r) {                // a[l] + .. + a[r]
        return queryRange(l, r, 0, n - 1, 0);
    }
};
