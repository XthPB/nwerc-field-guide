Fenwick tree

The original update sets a value and corrects the old contribution. For monotone prefix maxima implement point chmax and prefix max. A prefix maximum cannot be subtracted from another maximum.

From the ICPC Code Templates cheat sheet.

// USE: prefix sums with point updates: inversion count, "how many
//   values <= x so far", running sums. Shorter than a segment tree.
// NEEDS: 1-BASED indices 1..N.  TIME: O(log N) per call.
// Setup: BIT.assign(N + 1, 0); input.assign(N + 1, 0);
vector<long long> BIT;
vector<long long> input;               // current value of a[i]
void update(long long ind, long long value, long long N)
{                                      // SETS a[ind] = value
    for (long long i = ind; i <= N; i += (i & (-i))){
        BIT[i] += (value - input[ind]);
    }
    input[ind] = value;
}
long long query(long long l, long long r) {   // a[l] + .. + a[r]
    long long sum = 0;
    for (long long i = r; i > 0; i -= (i & (-i))) {
        sum += BIT[i];
    }
    for (long long i = l - 1; i > 0; i -= (i & (-i))) {
        sum -= BIT[i];
    }
    return sum;
}
// inversions: go left to right, ans += (#seen so far) - query(1, x),
//   then add 1 at x (values compressed to 1..N)
