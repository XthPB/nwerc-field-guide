Clip a polygon by a directed line

Keep the left side of A→B. Call this once for each counterclockwise edge of a convex clipping polygon. Points on the boundary are kept. The parameter eps is measured in signed cross-product units. Choose it for the coordinate scale, and use exact predicates when inputs permit.

struct ClipPoint {
    long double x, y;
};
std::vector<ClipPoint> clip_left(const std::vector<ClipPoint>& shape,
                                 ClipPoint a, ClipPoint b, long double eps) {
    auto side = [&](ClipPoint p) {
        long double v = (b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x);
        return std::abs(v) <= eps ? 0.0L : v;
    };
    std::vector<ClipPoint> result;
    if (shape.empty()) return result;
    ClipPoint p = shape.back();
    long double first = side(p);
    for (ClipPoint q : shape) {
        long double last = side(q);
        if ((first < 0 && last > 0) || (first > 0 && last < 0)) {
            long double t = first / (first - last);
            result.push_back({p.x + t * (q.x - p.x), p.y + t * (q.y - p.y)});
        }
        if (last >= 0) result.push_back(q);
        p = q;
        first = last;
    }
    return result;
}
