Disjoint-set union

This snippet merges ordinary components. A frontier DP additionally needs a canonical encoding of connectivity on its active boundary. A fresh temporary DSU can compute a transition.

From the ICPC Code Templates cheat sheet.

// USE: connectivity while ADDING edges (no deletions): components,
//   Kruskal MST, offline queries sorted by time/weight, cycle check.
// NEEDS: nodes 0..n-1.  TIME: ~O(1) amortised per call
//   (union by rank + path compression).
class dsu
{
private:
    vector<int> par;
    vector<int> rank;
public:
    dsu(int n)
    {
        for (int i = 0; i < n + 5; i++)
        {
            par.push_back(-1);
            rank.push_back(0);
        }
    }
    int findpar(int a)              // representative of a's component
    {
        if (par[a] == -1)
            return par[a] = a;
        if (par[a] == a)
            return a;
        return par[a] = findpar(par[a]);
    }
    void join(int a, int b) {       // merge the components of a and b
        if (findpar(a) == findpar(b))
            return;
        if (rank[findpar(a)] > rank[findpar(b)])
        {
            par[par[b]] = par[a];
            return;
        }
        else if (rank[findpar(a)] < rank[findpar(b)])
        {
            par[par[a]] = par[b];
            return;
        }
        par[par[a]] = par[b];
        rank[par[b]]++;
    }
};
// Component sizes: keep cnt[root], add when joining two roots.
