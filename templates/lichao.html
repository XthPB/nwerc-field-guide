Li Chao tree

The tree stores the lower envelope of integer affine functions. Derive slope, intercept and query coordinate from the recurrence. Bound the domain and evaluate products in a type that cannot overflow.

From the ICPC Code Templates cheat sheet.

// Li Chao tree (convex hull trick). USE: DP of the form
//   dp[i] = min over j < i of (m[j] * x[i] + c[j]):
//   for each i: query(x[i]), then addLine(m[i], c[i]).
// NEEDS: integer x in [lo, hi]; slopes/queries in ANY order;
//   a*x + b must fit in ll. TIME: O(log(hi - lo)) per call.
// maximum: addLine(-a, -b) and negate the query result.
struct LiChao {
    struct Line {
        ll a, b;
        ll at(ll x) const { return a * x + b; }
    };
    struct Node { Line line; int left = -1, right = -1; };
    vector<Node> t;
    ll lo, hi;
    LiChao(ll lo, ll hi) : lo(lo), hi(hi) {
        t.push_back({{0, LLONG_MAX}});
    }
    void addLine(ll a, ll b) { insert(0, lo, hi, {a, b}); }
    ll query(ll x) { return query(0, lo, hi, x); }  // LLONG_MAX: none
    int child(int node, bool goLeft) {   // create when missing
        int c = goLeft ? t[node].left : t[node].right;
        if (c != -1) return c;
        t.push_back({{0, LLONG_MAX}});
        c = t.size() - 1;
        if (goLeft) t[node].left = c; else t[node].right = c;
        return c;
    }
    void insert(int node, ll l, ll r, Line nw) {
        ll m = l + (r - l) / 2;
        bool leftBetter = nw.at(l) < t[node].line.at(l);
        bool midBetter = nw.at(m) < t[node].line.at(m);
        if (midBetter) swap(t[node].line, nw);  // keep winner at m
        if (l == r) return;
        if (leftBetter != midBetter)            // loser wins left
            insert(child(node, true), l, m, nw);
        else                                    // or right of m
            insert(child(node, false), m + 1, r, nw);
    }
    ll query(int node, ll l, ll r, ll x) {
        if (node == -1) return LLONG_MAX;
        ll m = l + (r - l) / 2;
        ll here = t[node].line.at(x);
        if (x <= m) return min(here, query(t[node].left, l, m, x));
        return min(here, query(t[node].right, m + 1, r, x));
    }
};
