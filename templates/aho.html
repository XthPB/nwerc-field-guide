Aho–Corasick automaton

Use automaton nodes as suffix states. Propagate forbidden or matching outputs through failure links before the DP. For a single short pattern a prefix-function automaton often suffices.

From the ICPC Code Templates cheat sheet.

// Aho-Corasick: search MANY patterns in a text at once.
// USE: #occurrences of every pattern, DP over texts that must avoid
//   banned words (dp[length][automaton state]).
// NEEDS: 'a'..'z'. TIME: O(26 * total pattern length) build,
//   O(text length) scan. MEMORY: 26 ints per node.
struct AhoCorasick {
    vector<array<int, 26>> nxt;   // after build(): full automaton
    vector<int> link, order;      // suffix link, BFS order of nodes
    AhoCorasick() { addNode(); }  // node 0 = root (empty string)
    int addNode() {
        nxt.push_back({});
        nxt.back().fill(-1);
        link.push_back(0);
        return nxt.size() - 1;
    }
    int insert(const string &p) {     // returns the node ending p
        int v = 0;
        for (char ch : p) {
            int c = ch - 'a';
            if (nxt[v][c] == -1) {
                int u = addNode();
                nxt[v][c] = u;
            }
            v = nxt[v][c];
        }
        return v;
    }
    void build() {                    // call after all inserts
        queue<int> q;
        q.push(0);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            order.push_back(v);
            for (int c = 0; c < 26; c++) {
                int u = nxt[v][c];
                int viaLink = v == 0 ? 0 : nxt[link[v]][c];
                if (u == -1) nxt[v][c] = viaLink;
                else { link[u] = viaLink; q.push(u); }
            }
        }
    }
    // cnt[node] = #times the text passes through a match of node
    vector<ll> countMatches(const string &text) {
        vector<ll> cnt(nxt.size(), 0);
        int v = 0;
        for (char ch : text) { v = nxt[v][ch - 'a']; cnt[v]++; }
        for (int i = order.size() - 1; i > 0; i--)
            cnt[link[order[i]]] += cnt[order[i]];
        return cnt;           // occurrences of p = cnt[insert(p) id]
    }
};
// banned words: bad[v] = (a word ends at v); in BFS order also
//   bad[v] |= bad[link[v]]; then never step into a bad state.
