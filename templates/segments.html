Segment and line predicates

// Segments and lines with the Point struct above. TIME: O(1).
// p on segment [a, b] (endpoints count)
bool onSegment(Point p, Point a, Point b) {
    return cross(a, b, p) == 0
        && min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x)
        && min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}
// do segments [a, b] and [c, d] share a point? (touching counts)
bool segmentsIntersect(Point a, Point b, Point c, Point d) {
    int d1 = sgn(cross(c, d, a)), d2 = sgn(cross(c, d, b));
    int d3 = sgn(cross(a, b, c)), d4 = sgn(cross(a, b, d));
    if (d1 * d2 < 0 && d3 * d4 < 0) return true;   // proper crossing
    return onSegment(a, c, d) || onSegment(b, c, d)
        || onSegment(c, a, b) || onSegment(d, a, b);
}
// intersection of infinite lines ab and cd, NOT parallel
// (parallel iff cross(b - a, d - c) == 0)
pair<long double, long double> lineIntersection(Point a, Point b,
                                                Point c, Point d) {
    long double t = (long double)cross(c - a, d - c)
                    / cross(b - a, d - c);
    return {a.x + t * (b.x - a.x), a.y + t * (b.y - a.y)};
}
long double distToSegment(Point p, Point a, Point b) {
    if (a == b || dot(p - a, b - a) <= 0) return sqrtl(dist2(p, a));
    if (dot(p - b, a - b) <= 0) return sqrtl(dist2(p, b));
    return fabsl((long double)cross(a, b, p)) / sqrtl(dist2(a, b));
}
