Lowest common ancestors

Use the common ancestor to compute tree distances or compress marked paths. A virtual tree also needs traversal order and ancestor tests.

From the ICPC Code Templates cheat sheet.

// LCA with binary lifting. USE: lowest common ancestor, distance
//   dist(u,v) = depth[u] + depth[v] - 2 depth[lca], k-th ancestor
//   (jump with up[v][i] for each bit i of k).
// NEEDS: connected TREE, nodes 0..n-1; set n and adj, then call
//   preprocess(root) ONCE (forest: add a new node joined to one node
//   of every tree and use it as root). TIME: O(n log n) build,
//   O(log n) per query. Recursive DFS (depth up to n).
// Globals clash with Bridges, Cut points, SCC: one per program.
int n, l;
vector<vector<int>> adj;
int timer;
vector<int> tin, tout, depth;
vector<vector<int>> up;
void dfs(int v, int p)
{
    tin[v] = ++timer;
    depth[v] = (v == p) ? 0 : depth[p] + 1;
    up[v][0] = p;
    for (int i = 1; i <= l; ++i)
        up[v][i] = up[up[v][i-1]][i-1];
    for (int u : adj[v]) {
        if (u != p)
            dfs(u, v);
    }
    tout[v] = ++timer;
}
bool is_ancestor(int u, int v)
{
    return tin[u] <= tin[v] && tout[u] >= tout[v];
}
int lca(int u, int v)
{
    if (is_ancestor(u, v))
        return u;
    if (is_ancestor(v, u))
        return v;
    for (int i = l; i >= 0; --i) {
        if (!is_ancestor(up[u][i], v))
            u = up[u][i];
    }
    return up[u][0];
}
void preprocess(int root) {
    tin.resize(n);
    tout.resize(n);
    depth.resize(n);
    timer = 0;
    l = ceil(log2(n));
    up.assign(n, vector<int>(l + 1));
    dfs(root, root);
}
int dist(int u, int v) {
    return depth[u] + depth[v] - 2 * depth[lca(u, v)];
}
