2D Fenwick tree for point deltas and rectangle sums

Use zero-based cells and half-open rectangles [top,bottom) × [left,right). Add +1 or −1 when a mixed difference switches between zero and nonzero. Empty rectangles sum to zero. The dense tree uses O(rows × columns) space.

struct FenwickGrid {
    int rows, columns;
    std::vector<std::vector<long long>> bit;
    FenwickGrid(int r, int c) : rows(r), columns(c), bit(r+1, std::vector<long long>(c+1)) {}
    void add(int row, int column, long long delta) {
        for (int i = row+1; i <= rows; i += i & -i)
            for (int j = column+1; j <= columns; j += j & -j)
                bit[i][j] += delta;
    }
    long long prefix(int bottom, int right) const {
        long long result = 0;
        for (int i = bottom; i > 0; i -= i & -i)
            for (int j = right; j > 0; j -= j & -j)
                result += bit[i][j];
        return result;
    }
    long long sum(int top, int left, int bottom, int right) const {
        return prefix(bottom,right) - prefix(top,right)
             - prefix(bottom,left) + prefix(top,left);
    }
};
