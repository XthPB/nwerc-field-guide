Pollard rho

Use a primality test as the recursion base, wide modular multiplication, and retry unsuccessful walks.

From the ICPC Code Templates section “Pollard Rho (factorise up to 1e18)”. Use the code with the types and conventions shown in the excerpt.

Include these supporting sections first · Miller–Rabin primality testing

// Pollard rho: factorise n up to ~1e18 (trial division needs 1e9
//   steps). NEEDS: isPrime from Miller-Rabin. TIME: ~n^(1/4) steps
//   per factor (a few ms for 1e18). Factors come out UNSORTED.
mt19937_64 rngRho(12345);
ll rho(ll n) {                       // some divisor 1 < d < n
    if (n % 2 == 0) return 2;
    while (true) {
        ll c = rngRho() % (n - 1) + 1, x = rngRho() % n, y = x, d = 1;
        auto f = [&](ll v) { return (ll)(((__int128)v * v + c) % n); };
        while (d == 1) {
            x = f(x);
            y = f(f(y));
            d = __gcd(x > y ? x - y : y - x, n);
        }
        if (d != n) return d;        // else retry with a new c
    }
}
void factorRho(ll n, vector<ll> &out) {   // primes, with repeats
    if (n == 1) return;
    if (isPrime(n)) { out.push_back(n); return; }
    ll d = rho(n);
    factorRho(d, out);
    factorRho(n / d, out);
}
