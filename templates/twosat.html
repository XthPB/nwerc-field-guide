2-SAT

Check the SCC numbering convention before turning an order into a Boolean assignment.

// 2-SAT. USE: n boolean choices with constraints on PAIRS
//   ("a or b", "not both", "a implies b", "a != b").
// NEEDS: the SCC code above. TIME: O(n + #clauses).
// variables 0..n-1; node 2x = "x true", node 2x+1 = "x false"
struct TwoSat {
    int n;
    vector<vector<int>> adj;
    TwoSat(int vars) : n(vars), adj(2 * vars) {}
    int node(int x, bool value) { return 2 * x + (value ? 0 : 1); }
    // clause: (x_a == va) OR (x_b == vb)
    void addOr(int a, bool va, int b, bool vb) {
        adj[node(a, !va)].push_back(node(b, vb));  // not A -> B
        adj[node(b, !vb)].push_back(node(a, va));  // not B -> A
    }
    void addImplies(int a, bool va, int b, bool vb) {
        addOr(a, !va, b, vb);
    }
    void forceValue(int a, bool va) { addOr(a, va, a, va); }
    vector<int> solve() {      // value of every variable, {} if none
        vector<vector<int>> comps, cond;
        strongly_connected_components(adj, comps, cond);
        vector<int> compId(2 * n);
        for (int i = 0; i < (int)comps.size(); i++)
            for (int v : comps[i]) compId[v] = i;
        vector<int> value(n);
        for (int x = 0; x < n; x++) {
            if (compId[2 * x] == compId[2 * x + 1]) return {};
            value[x] = compId[2 * x] > compId[2 * x + 1];
        }
        return value;
    }
};
// a != b: addOr(a,1,b,1), addOr(a,0,b,0).  not both: addOr(a,0,b,0)
