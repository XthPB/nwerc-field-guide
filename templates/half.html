Meet in the middle for sums

Keep both lists including the empty subset. For an exact match use equal_range so repeated sums retain their multiplicity. For a closest partner inspect both neighbours of lower_bound.

From the ICPC Code Templates cheat sheet.

// Meet in the middle. USE: n <= ~40 items where 2^n is too slow but
//   2^(n/2) is fine: subset sums, n = 40 knapsack, 4-SUM (pairs of
//   pairs), and bidirectional BFS (search from start AND goal).
// IDEA: split the items into two halves, list everything for each
//   half, sort one list, and for each entry of the other list binary
//   search the best partner. TIME: O(2^(n/2) * n).
vector<ll> subsetSums(const vector<ll> &a) {    // all 2^|a| sums
    vector<ll> sums = {0};
    for (ll x : a) {
        int k = sums.size();
        for (int i = 0; i < k; i++) sums.push_back(sums[i] + x);
    }
    return sums;
}
ll countSubsetsAtMost(const vector<ll> &a, ll S) {  // #subsets sum<=S
    int h = a.size() / 2;
    vector<ll> L = subsetSums(vector<ll>(a.begin(), a.begin() + h));
    vector<ll> R = subsetSums(vector<ll>(a.begin() + h, a.end()));
    sort(R.begin(), R.end());
    ll count = 0;
    for (ll x : L)
        count += upper_bound(R.begin(), R.end(), S - x) - R.begin();
    return count;
}
// sum EXACTLY S: equal_range instead. CLOSEST to S: lower_bound and
//   the element before it.
