Centroid decomposition

For path counts, query earlier child branches before inserting distances from the current branch. Recurse after removing the centroid so that other paths are attributed exactly once.

From the ICPC Code Templates cheat sheet.

// Centroid decomposition. USE: questions about ALL PATHS of a tree
//   (count paths of length k, paths with sum <= K), "distance to
//   the nearest marked node" with updates.
// IDEA: every path u-v passes through the highest centroid c that
//   lies above both u and v; each node has <= log2(n) + 1 centroids
//   above it. anc[v] = list of {centroid c above v, dist(v, c)}.
// NEEDS: tree, nodes 0..n-1. TIME: O(n log n) build and memory.
struct Centroid {
    vector<vector<int>> adj;
    vector<bool> removed;
    vector<int> par, sub, dist, cpar;      // cpar: centroid-tree parent
    vector<vector<pair<int, int>>> anc;
    Centroid(const vector<vector<int>> &g) : adj(g) {
        int n = g.size();
        removed.assign(n, false);
        par.assign(n, -1); sub.assign(n, 0); dist.assign(n, 0);
        cpar.assign(n, -1); anc.assign(n, {});
        decompose(0, -1);
    }
    vector<int> piece(int s) {   // BFS over the not-removed part of s
        vector<int> order = {s};
        par[s] = -1;
        dist[s] = 0;
        for (int i = 0; i < (int)order.size(); i++) {
            int u = order[i];
            for (int v : adj[u])
                if (v != par[u] && !removed[v]) {
                    par[v] = u;
                    dist[v] = dist[u] + 1;
                    order.push_back(v);
                }
        }
        return order;
    }
    void decompose(int s, int above) {   // recursion depth O(log n)
        vector<int> order = piece(s);
        int total = order.size();
        for (int i = total - 1; i >= 0; i--) {
            int u = order[i];
            sub[u] = 1;
            for (int v : adj[u])
                if (v != par[u] && !removed[v]) sub[u] += sub[v];
        }
        int c = s;                       // walk to the centroid
        while (true) {
            int next = -1;
            for (int v : adj[c])
                if (v != par[c] && !removed[v] && 2 * sub[v] > total)
                    next = v;
            if (next == -1) break;
            c = next;
        }
        cpar[c] = above;
        for (int u : piece(c)) anc[u].push_back({c, dist[u]});
        removed[c] = true;
        for (int v : adj[c])
            if (!removed[v]) decompose(v, c);
    }
};
// nearest marked node: best[c] = INF for every c, then
//   mark(v):  for (auto [c, d] : anc[v]) best[c] = min(best[c], d);
//   query(v): min over (c, d) in anc[v] of best[c] + d.
