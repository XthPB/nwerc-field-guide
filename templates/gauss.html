Gaussian elimination

Put one row per expectation equation and one column per unknown state. Handle absorbing states separately. A tridiagonal row system admits a cheaper specialized solver.

From the ICPC Code Templates cheat sheet.

// Gaussian elimination: solve A x = b (n equations, m unknowns).
// USE: expected values / probabilities when states form CYCLES
//   (random walks): E[v] - sum p(v,u) E[u] = cost(v); circuits.
// TIME: O(n m^2) (~0.1 s for 500 x 500). Doubles, partial pivoting.
// returns 0 = no solution, 1 = unique, 2 = infinitely many
int gauss(vector<vector<double>> a, vector<double> b,
          vector<double> &x) {
    const double eps = 1e-9;
    int n = a.size(), m = a[0].size();
    for (int i = 0; i < n; i++) a[i].push_back(b[i]);
    vector<int> where(m, -1);      // where[col] = its pivot row
    int row = 0;
    for (int col = 0; col < m && row < n; col++) {
        int best = row;
        for (int i = row; i < n; i++)
            if (fabs(a[i][col]) > fabs(a[best][col])) best = i;
        if (fabs(a[best][col]) < eps) continue;
        swap(a[best], a[row]);
        where[col] = row;
        for (int i = 0; i < n; i++) {
            if (i == row) continue;
            double factor = a[i][col] / a[row][col];
            for (int j = col; j <= m; j++)
                a[i][j] -= factor * a[row][j];
        }
        row++;
    }
    x.assign(m, 0);
    for (int j = 0; j < m; j++)
        if (where[j] != -1) x[j] = a[where[j]][m] / a[where[j]][j];
    for (int i = 0; i < n; i++) {
        double sum = 0;
        for (int j = 0; j < m; j++) sum += a[i][j] * x[j];
        if (fabs(sum - a[i][m]) > 1e-6) return 0;
    }
    for (int j = 0; j < m; j++)
        if (where[j] == -1) return 2;
    return 1;
}
