Trie of strings

Starting from a reachable split position, traverse matching dictionary prefixes, and relax the end position at every terminal. Preserve word multiplicity if separately labeled dictionary entries matter.

From the ICPC Code Templates cheat sheet.

// Trie of words (Iancu's). USE: #words equal to w, #words starting
//   with prefix w, longest common prefix, word games.
// NEEDS: 'a'..'z' only. TIME: O(|w|) per call. MEMORY: 26 pointers
//   per node (total length up to ~1e6 is fine).
// cp = #words passing through a node, cw = #words ending there.
// erase(w) only for words that are present (check
//   countWordsEqualTo(w) > 0 first), else the counts go wrong.
class Trie
{
    struct Node
    {
        Node *arr[26] = {};   // FIXED: "= {}" (children must start NULL)
        int cp = 0;
        int cw = 0;
    };
public:
    Node *root;
    Node *temp;
    Trie()
    {
        root = new Node;
    }
    void insert(string &word)
    {
        temp = root;
        int n = word.length();
        for (int i = 0; i < n; i++)
        {
            if (temp->arr[word[i] - 'a'] == NULL)
            {
                temp->arr[word[i] - 'a'] = new Node;
            }
            temp = temp->arr[word[i] - 'a'];
            temp->cp = temp->cp + 1;
        }
        temp->cw = temp->cw + 1;
    }
    int countWordsEqualTo(string &word)
    {
        temp = root;
        int n = word.length();
        for (int i = 0; i < n; i++)
        {
            if (temp->arr[word[i] - 'a'] == NULL)
            {
                return 0;
            }
            temp = temp->arr[word[i] - 'a'];
        }
        return temp->cw;
    }
    int countWordsStartingWith(string &word)
    {
        temp = root;
        int n = word.length();
        for (int i = 0; i < n; i++)
        {
            if (temp->arr[word[i] - 'a'] == NULL)
            {
                return 0;
            }
            temp = temp->arr[word[i] - 'a'];
        }
        return temp->cp;
    }
    void erase(string &word)
    {
        temp = root;
        int n = word.length();
        for (int i = 0; i < n; i++)
        {
            if (temp->arr[word[i] - 'a'] == NULL)
            {
                temp->arr[word[i] - 'a'] = new Node;
            }
            temp = temp->arr[word[i] - 'a'];
            temp->cp = temp->cp - 1;
        }
        temp->cw = temp->cw - 1;
    }
};
