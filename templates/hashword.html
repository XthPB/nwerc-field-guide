String fingerprints

This guide adaptation handles an empty pattern explicitly. The fingerprint encoding assumes lowercase English letters. Verify candidate matches when exact equality is required.

Adapted for this guide from the ICPC Code Templates section “Rabin-Karp”. Use the code with the types and conventions shown in the excerpt.

vector<int> rabin_karp(string const &pattern, string const &text) {
    int n = pattern.size();
    int m = text.size();
    if (n == 0) {
        vector<int> boundaries(m + 1);
        iota(boundaries.begin(), boundaries.end(), 0);
        return boundaries;
    }
    if (n > m)
        return {};
    constexpr int P = 37;
    constexpr int MOD = 1e9 + 7;
    vector<long long> p_pow(m);
    p_pow[0] = 1;
    for (int i = 1; i < m; i++)
        p_pow[i] = (p_pow[i - 1] * P) % MOD;
    vector<long long> h(m + 1);
    h[0] = 0;
    for (int i = 0; i < m; i++)
        h[i + 1] = (h[i] + (text[i] - 'a' + 1) * p_pow[i]) % MOD;
    long long h_p = 0;
    for (int i = 0; i < n; i++)
        h_p = (h_p + (pattern[i] - 'a' + 1) * p_pow[i]) % MOD;
    vector<int> indices;
    for (int i = 0; i + n - 1 < m; i++) {
        long long temp_h = (h[i + n] + MOD - h[i]) % MOD;
        if (temp_h == h_p * p_pow[i] % MOD)
            indices.push_back(i);
    }
    return indices;
}
