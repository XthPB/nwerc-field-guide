Manacher palindromic radii

Odd and even centers have different index conventions.

From the ICPC Code Templates section “Manacher, old heading "Z Algorithm" (Iancu)”. Use the code with the types and conventions shown in the excerpt.

// Manacher (Iancu's; the old heading said "Z Algorithm").
// USE: all palindromes in O(n): longest palindromic substring,
//   #palindromic substrings, "is s[l..r] a palindrome" in O(1).
// P[i] = radius in "#s0#s1#...#" = length of the longest palindrome
//   centred at i in the ORIGINAL string. 0-based inclusive [l, r].
// longest = max P[i]; #palindromic substrings = sum of (P[i] + 1) / 2
vector<int> preprocess(const string& s) {
    string processed_string = "#";
    for(char c : s) {
        processed_string += c;
        processed_string += '#';
    }
    int n = processed_string.size();
    vector<int> P(n, 0);
    int C = 0, R = 0;
    for (int i = 1; i < n; ++i) {
        int mirror = 2 * C - i;
        if (i < R)
            P[i] = min(R - i, P[mirror]);
        while (i - (1 + P[i]) >= 0 && i + (1 + P[i]) < n &&
               processed_string[i - (1 + P[i])] ==
               processed_string[i + (1 + P[i])])
            P[i]++;
        if (i + P[i] > R) {
            C = i;
            R = i + P[i];
        }
    }
    return P;
}
bool is_palindrome_substring(const string& s, int l, int r,
                             const vector<int>& P) {
    // FIXED (old line was wrong for ~1/3 of substrings):
    // return P[(l + r) / 2] >= (r - l + 1) / 2;
    return P[l + r + 1] >= r - l + 1;
}
