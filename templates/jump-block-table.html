Forward jumps with one editable block

For positive jumps, query returns the last zero-based index and the number of jumps up to and including the exit. Rebuild one block after a point edit. The destination is capped at n using wide addition.

struct ForwardJumps {
    int n, width;
    std::vector<int> jump, exit, hops, last;
    ForwardJumps(std::vector<int> initial, int blockWidth)
        : n((int)initial.size()), width(std::max(1, blockWidth)), jump(std::move(initial)),
          exit(n), hops(n), last(n) {
        for (int lo = 0; lo < n; lo += width) rebuild(lo);
    }
    void rebuild(int lo) {
        int end = std::min(n,lo+width);
        for (int i = end-1; i >= lo; --i) {
            int to = (int)std::min<long long>(n, (long long)i+jump[i]);
            if (to >= end) {
                exit[i] = to; hops[i] = 1; last[i] = i;
            } else {
                exit[i] = exit[to]; hops[i] = 1+hops[to]; last[i] = last[to];
            }
        }
    }
    void set(int i, int value) {
        jump[i] = value;
        rebuild((i/width)*width);
    }
    std::pair<int,int> query(int i) const {
        int total = 0, end = i;
        while (i < n) {
            total += hops[i];
            end = last[i];
            i = exit[i];
        }
        return {end,total};
    }
};
