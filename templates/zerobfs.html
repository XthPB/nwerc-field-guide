0–1 BFS

Push zero-cost improvements to the front and cost-one improvements to the back.

From the ICPC Code Templates section “0-1 BFS”. Use the code with the types and conventions shown in the excerpt.

// 0-1 BFS: shortest paths when every weight is 0 or 1 (e.g. "free
//   move vs. costly move" on grids). TIME: O(V + E), no heap.
vector<int> bfs01(int src,
                  const vector<vector<pair<int, int>>> &adj) {
    vector<int> dist(adj.size(), INT_MAX);
    deque<int> dq;
    dist[src] = 0;
    dq.push_back(src);
    while (!dq.empty()) {
        int u = dq.front();
        dq.pop_front();
        for (auto [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                if (w == 0) dq.push_front(v);
                else dq.push_back(v);
            }
        }
    }
    return dist;
}
// all weights 1: plain BFS. grid: cell (r, c) -> id r * W + c
// int dr[] = {-1, 1, 0, 0}, dc[] = {0, 0, -1, 1};
