Sparse table

Overlapping blocks are valid for idempotent operators such as minimum.

From the ICPC Code Templates section “Sparse Table”. Use the code with the types and conventions shown in the excerpt.

// Sparse table: STATIC array (no updates), O(1) range query.
// USE: range min/max/gcd/and/or on a fixed array, LCA via Euler tour,
//   LCP of two suffixes (min over the lcp array).
// ONLY for idempotent ops (op(x, x) = x): NOT for sum or xor.
// TIME: build O(n log n), query O(1).  MEMORY: O(n log n).
template<class T>
struct SparseTable {
    int n, K;
    vector<vector<T>> st;
    vector<int> lg;
    SparseTable(const vector<T>& a) {
        n = a.size();
        K = 32 - __builtin_clz(n);
        st.assign(K, vector<T>(n));
        lg.resize(n + 1);
        for (int i = 2; i <= n; i++)
            lg[i] = lg[i / 2] + 1;
        st[0] = a;
        for (int k = 1; k < K; k++)
            for (int i = 0; i + (1 << k) <= n; i++)
                st[k][i] = merge(st[k-1][i],
                                 st[k-1][i + (1 << (k-1))]);
    }
    T merge(T a, T b) {
        return min(a, b); // change operation here
    }
    T query(int l, int r) {             // inclusive [l, r], 0-based
        int k = lg[r - l + 1];
        return merge(st[k][l],
                     st[k][r - (1 << k) + 1]);
    }
};
