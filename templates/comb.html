Binomial coefficients and modular arithmetic

Use modular binomial coefficients for independent choices and unrestricted grid paths. Check that the modulus and factorial range make the inverses used by the template valid.

From the ICPC Code Templates cheat sheet.

// USE: counting answers "modulo 1e9+7", nCr, probabilities as p/q.
// NEEDS: MOD prime for inverse(). TIME: power O(log e),
//   initFactorials O(n), nCr O(1).  a * b < 1e18 so ll is safe.
const ll MOD = 1e9 + 7;            // or 998244353
ll power(ll base, ll exp, ll mod = MOD) {
    base %= mod;
    if (base < 0) base += mod;
    ll result = 1;
    while (exp > 0) {
        if (exp & 1) result = result * base % mod;
        base = base * base % mod;
        exp >>= 1;
    }
    return result;
}
ll inverse(ll a) { return power(a, MOD - 2); }   // a % MOD != 0
vector<ll> fact, invFact;
void initFactorials(int n) {       // call once with the largest n
    fact.assign(n + 1, 1);
    invFact.assign(n + 1, 1);
    for (int i = 1; i <= n; i++) fact[i] = fact[i - 1] * i % MOD;
    invFact[n] = inverse(fact[n]);
    for (int i = n; i > 0; i--) invFact[i - 1] = invFact[i] * i % MOD;
}
ll nCr(int n, int r) {
    if (r < 0 || r > n) return 0;
    return fact[n] * invFact[r] % MOD * invFact[n - r] % MOD;
}
// "p/q mod MOD" = p % MOD * inverse(q % MOD) % MOD
// negative x: ((x % MOD) + MOD) % MOD
