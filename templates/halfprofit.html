Meet in the middle for profit

Use the prefix-best pass on the weight-sorted right list. When reconstructing a choice, update its stored witness whenever its prefix profit improves. This snippet assumes nonnegative weights.

From the ICPC Code Templates cheat sheet.

// n = 40 knapsack: best value with weight <= W
ll knapsackMitm(const vector<pair<ll, ll>> &items, ll W) {  // {w, v}
    int h = items.size() / 2;
    auto half = [&](int from, int to) {        // all {weight, value}
        vector<pair<ll, ll>> res = {{0, 0}};
        for (int i = from; i < to; i++) {
            int k = res.size();
            for (int j = 0; j < k; j++)
                res.push_back({res[j].first + items[i].first,
                               res[j].second + items[i].second});
        }
        return res;
    };
    vector<pair<ll, ll>> L = half(0, h), R = half(h, items.size());
    sort(R.begin(), R.end());
    for (int i = 1; i < (int)R.size(); i++)    // best value up to here
        R[i].second = max(R[i].second, R[i - 1].second);
    ll best = 0;
    for (auto [w, v] : L) {
        if (w > W) continue;
        auto it = upper_bound(R.begin(), R.end(),
                              make_pair(W - w, LLONG_MAX));
        best = max(best, v + prev(it)->second);  // R[0] = {0, 0}
    }
    return best;
}
