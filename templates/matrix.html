Matrix multiplication and power

The sheet multiplies with ordinary modular addition and multiplication. For minimum-cost transitions replace those by min and plus, use an infinite unreachable value, and skip unreachable additions.

From the ICPC Code Templates cheat sheet.

// Matrix power. USE: linear recurrences with huge n (Fibonacci-like,
//   f(n) = c1 f(n-1) + .. + ck f(n-k)), #walks of length k in a graph.
// NEEDS: MOD from above. TIME: O(k^3 log n) for a k x k matrix.
typedef vector<vector<ll>> Matrix;
Matrix multiply(const Matrix &A, const Matrix &B) {
    int n = A.size(), inner = B.size(), m = B[0].size();
    Matrix C(n, vector<ll>(m, 0));
    for (int i = 0; i < n; i++)
        for (int k = 0; k < inner; k++)
            for (int j = 0; j < m; j++)
                C[i][j] = (C[i][j] + A[i][k] * B[k][j]) % MOD;
    return C;
}
Matrix matrixPower(Matrix A, ll exp) {
    int n = A.size();
    Matrix result(n, vector<ll>(n, 0));
    for (int i = 0; i < n; i++) result[i][i] = 1;
    while (exp > 0) {
        if (exp & 1) result = multiply(result, A);
        A = multiply(A, A);
        exp >>= 1;
    }
    return result;
}
// Fibonacci: {{1,1},{1,0}}^n = {{F(n+1),F(n)},{F(n),F(n-1)}}
// (adjacency matrix)^k [u][v] = #walks of length k from u to v
