Strongly connected components

Build a condensation DAG and define what one component contributes. A simple path with restricted entry and exit may need its own internal subproblem.

From the ICPC Code Templates cheat sheet.

// Strongly connected components (Kosaraju).
// USE: directed graphs with cycles: merge each cycle-group into one
//   node, then DP on the condensation DAG; 2-SAT; "which nodes can
//   reach each other".  NEEDS: DIRECTED, nodes 0..n-1.
// TIME: O(V + E), recursive.  components come out in TOPOLOGICAL
//   order of the condensation (components[0] has no incoming edge).
// Globals dfs/visited clash with Bridges, Cut points, LCA.
vector<bool> visited; // which vertices are already visited
// DFS from v; each vertex is appended to output when dfs leaves it
void dfs(int v, vector<vector<int>> const& adj, vector<int> &output) {
    visited[v] = true;
    for (auto u : adj[v])
        if (!visited[u])
            dfs(u, adj, output);
    output.push_back(v);
}
// output: components -- the strongly connected components in G
// output: adj_cond -- adjacency of the condensation, indexed by the
//   root (smallest vertex) of each component; may contain duplicates
void strongly_connected_components(vector<vector<int>> const& adj,
                                  vector<vector<int>> &components,
                                  vector<vector<int>> &adj_cond) {
    int n = adj.size();
    components.clear(), adj_cond.clear();
    vector<int> order; // vertices sorted by exit time
    visited.assign(n, false);
    // first series of depth first searches
    for (int i = 0; i < n; i++)
        if (!visited[i])
            dfs(i, adj, order);
    // create adjacency list of G^T
    vector<vector<int>> adj_rev(n);
    for (int v = 0; v < n; v++)
        for (int u : adj[v])
            adj_rev[u].push_back(v);
    visited.assign(n, false);
    reverse(order.begin(), order.end());
    vector<int> roots(n, 0); // root vertex of each vertex's SCC
    // second series of depth first searches
    for (auto v : order)
        if (!visited[v]) {
            std::vector<int> component;
            dfs(v, adj_rev, component);
            components.push_back(component);
            int root = *min_element(begin(component), end(component));
            for (auto u : component)
                roots[u] = root;
        }
    // add edges to condensation graph
    adj_cond.assign(n, {});
    for (int v = 0; v < n; v++)
        for (auto u : adj[v])
            if (roots[v] != roots[u])
                adj_cond[roots[v]].push_back(roots[u]);
}
// condensation: min #start nodes reaching all = #sources (in-degree 0)
//   min #edges to add to make it strongly connected = max(#sources,
//   #sinks), 0 if one SCC. "needs all it reaches": best is a sink SCC
