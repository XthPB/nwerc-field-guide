Monotone stack and sliding deque

The sheet includes nearest-smaller and sliding minimum examples. Reverse comparisons for maximum. Track indices so expiration still works for equal scores.

From the ICPC Code Templates cheat sheet.

// Monotonic stack / deque. USE: "nearest smaller/greater to the
//   left/right", largest rectangle in a histogram, max of every
//   window, DP of the form dp[i] = min(dp[j]) + c over a SLIDING
//   range of j.  TIME: O(n) total (each index pushed/popped once).
// left[i] = largest j < i with a[j] < a[i], or -1
vector<int> prevSmaller(const vector<ll> &a) {
    int n = a.size();
    vector<int> left(n), st;       // st: indices, values increase
    for (int i = 0; i < n; i++) {
        while (!st.empty() && a[st.back()] >= a[i]) st.pop_back();
        left[i] = st.empty() ? -1 : st.back();
        st.push_back(i);
    }
    return left;
}
// next smaller: use the reversed array. Greater: <= in the while.
ll largestRectangle(const vector<ll> &h) {    // histogram
    int n = h.size();
    vector<int> left = prevSmaller(h);
    vector<ll> rev(h.rbegin(), h.rend());
    vector<int> rightRev = prevSmaller(rev);
    ll best = 0;
    for (int i = 0; i < n; i++) {
        int right = n - 1 - rightRev[n - 1 - i];  // n if none
        best = max(best, h[i] * (right - left[i] - 1));
    }
    return best;
}
// minimum of every window a[i-k+1..i] (max: flip the >=)
vector<ll> slidingMin(const vector<ll> &a, int k) {
    deque<int> dq;          // indices, values increase front->back
    vector<ll> result;
    for (int i = 0; i < (int)a.size(); i++) {
        while (!dq.empty() && a[dq.back()] >= a[i]) dq.pop_back();
        dq.push_back(i);
        if (dq.front() <= i - k) dq.pop_front();
        if (i >= k - 1) result.push_back(a[dq.front()]);
    }
    return result;
}
