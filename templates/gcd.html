Extended gcd and residues

Positive gcd and normalized residues simplify divisibility tests.

From the ICPC Code Templates section “Extended GCD + Chinese Remainder (any moduli)”. Use the code with the types and conventions shown in the excerpt.

// USE: a*x + b*y = g, inverse mod a NON-prime m, systems
//   "x = r1 mod m1, x = r2 mod m2" (cycles that must line up).
// TIME: O(log min(a, b)).
ll extgcd(ll a, ll b, ll &x, ll &y) {   // returns g = gcd(a, b)
    if (b == 0) { x = 1; y = 0; return a; }
    ll x1, y1;
    ll g = extgcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return g;
}
// a^-1 mod m for ANY m (prime or not); -1 if gcd(a, m) != 1
ll inverseMod(ll a, ll m) {
    ll x, y;
    if (extgcd(((a % m) + m) % m, m, x, y) != 1) return -1;
    return ((x % m) + m) % m;
}
// a*x + b*y = c: solvable iff c % g == 0; x0*(c/g), y0*(c/g) is
// one solution, all: x + k*(b/g), y - k*(a/g)
// CRT: m1, m2 need NOT be coprime. Returns {x, lcm} with
// 0 <= x < lcm, or {-1, -1} if impossible. lcm must fit in ll.
pair<ll, ll> crt(ll r1, ll m1, ll r2, ll m2) {
    ll p, q;
    ll g = extgcd(m1, m2, p, q);
    if ((r2 - r1) % g != 0) return {-1, -1};
    ll lcm = m1 / g * m2;
    __int128 k = (__int128)((r2 - r1) / g) * p % (m2 / g);
    __int128 x = (r1 + k * m1) % lcm;
    if (x < 0) x += lcm;
    return {(ll)x, lcm};
}
