Find the first position exceeding a threshold

Store a maximum in each node. Starting at an inclusive zero-based index, descend left first and skip intervals whose maximum cannot exceed the threshold. Set a column’s value to its largest active y. Use the lowest representable integer for an empty column when queries use the same integer type.

struct FirstAbove {
    using Value = long long;
    int count, base = 1;
    std::vector<Value> maximum;
    explicit FirstAbove(int n) : count(n) {
        while (base < n) base *= 2;
        maximum.assign(2 * base, std::numeric_limits<Value>::lowest());
    }
    void set(int pos, Value value) {
        int i = base + pos;
        maximum[i] = value;
        while (i > 1) {
            i /= 2;
            maximum[i] = std::max(maximum[2*i], maximum[2*i+1]);
        }
    }
    int first(int start, Value threshold) const {
        return descend(1, 0, base, start, threshold);
    }
private:
    int descend(int v, int lo, int hi, int start, Value threshold) const {
        if (hi <= start || lo >= count || maximum[v] <= threshold) return -1;
        if (hi - lo == 1) return lo;
        int mid = lo + (hi - lo)/2;
        int left = descend(2*v, lo, mid, start, threshold);
        if (left != -1) return left;
        return descend(2*v+1, mid, hi, start, threshold);
    }
};
