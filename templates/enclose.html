Minimum enclosing circle

// Minimum enclosing circle (randomised, expected O(n)).
// NEEDS: ld and PointD from Circle Intersections, >= 1 point.
// returns {centre, radius}
PointD circumCentre(PointD a, PointD b, PointD c) {
    ld bx = b.x - a.x, by = b.y - a.y, cx = c.x - a.x, cy = c.y - a.y;
    ld d = 2 * (bx * cy - by * cx);
    ld b2 = bx * bx + by * by, c2 = cx * cx + cy * cy;
    return {a.x + (cy * b2 - by * c2) / d,
            a.y + (bx * c2 - cx * b2) / d};
}
pair<PointD, ld> minEnclosingCircle(vector<PointD> p) {
    shuffle(p.begin(), p.end(), mt19937(7));
    auto dist = [](PointD a, PointD b) {
        return hypotl(a.x - b.x, a.y - b.y);
    };
    auto outside = [&](PointD c, ld r, PointD q) {
        return dist(c, q) > r * (1 + 1e-12) + 1e-12;
    };
    PointD c = p[0];
    ld r = 0;
    for (int i = 0; i < (int)p.size(); i++) {
        if (!outside(c, r, p[i])) continue;
        c = p[i], r = 0;                       // p[i] on the boundary
        for (int j = 0; j < i; j++) {
            if (!outside(c, r, p[j])) continue;
            c = {(p[i].x + p[j].x) / 2, (p[i].y + p[j].y) / 2};
            r = dist(c, p[i]);                 // p[i], p[j] on it
            for (int k = 0; k < j; k++) {
                if (!outside(c, r, p[k])) continue;
                c = circumCentre(p[i], p[j], p[k]);
                r = dist(c, p[i]);
            }
        }
    }
    return {c, r};
}
