Rooted tree DP and rerooting

Use reverse traversal for bottom-up child messages. Use forward traversal for the parent contribution. Derive a new message when the objective differs from distance or subtree size.

From the ICPC Code Templates cheat sheet.

// Tree DP without recursion + rerooting.
// USE: rooted-tree DP on deep trees (a path of 1e6 nodes would
//   overflow a recursive DFS); "answer for every node as the root"
//   in O(n) instead of O(n^2).  NEEDS: tree, nodes 0..n-1.
// rootTree: BFS order from root. Loop order BACKWARDS = children
//   before parents (subtree sums); FORWARDS = parents first.
void rootTree(int root, const vector<vector<int>> &adj,
              vector<int> &parent, vector<int> &order) {
    parent.assign(adj.size(), -1);
    order = {root};
    for (int i = 0; i < (int)order.size(); i++) {
        int u = order[i];
        for (int v : adj[u])
            if (v != parent[u]) { parent[v] = u; order.push_back(v); }
    }
}
// Example: answer[v] = sum of distances from v to all nodes, O(n)
vector<ll> sumOfDistances(const vector<vector<int>> &adj) {
    int n = adj.size();
    vector<int> parent, order;
    rootTree(0, adj, parent, order);
    vector<ll> sub(n, 1);          // subtree size
    vector<ll> down(n, 0);         // sum of distances into subtree
    for (int i = n - 1; i > 0; i--) {
        int v = order[i], p = parent[v];
        sub[p] += sub[v];
        down[p] += down[v] + sub[v];
    }
    vector<ll> answer(n);
    answer[0] = down[0];
    for (int i = 1; i < n; i++) {  // move root from p to child v:
        int v = order[i], p = parent[v];
        // sub[v] nodes get 1 closer, the other n - sub[v] 1 farther
        answer[v] = answer[p] - sub[v] + (n - sub[v]);
    }
    return answer;
}
// general: down[] bottom-up, then up[v] = parent's value minus v's
// part (max/min: use prefix and suffix maxima over children)
// diameter: a = farthest from any node, b = farthest from a
// WEIGHTED: node weights (W = total, sub[v] = subtree weight), edge
//   (p,v) of length L: answer[v] = answer[p] + L * (W - 2 * sub[v]).
// best node (min sum w * dist) = weighted median: from the root, step
//   into the child with 2 * sub[child] > W while one exists.
// tree knapsack dp[v][k], k <= sub[v]: merge child c by nw[i + j] +=
//   dp[v][i] * dp[c][j], i <= size merged so far, j <= sub[c]: O(n^2)
//   total (each pair meets once, at its LCA). Cap k at K: O(n K).
