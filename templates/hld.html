// Heavy-light decomposition: PATH and SUBTREE queries with updates.
// USE: "change the value of node v", "sum / max on the path u-v",
//   "sum of the subtree of v".
// NEEDS: tree, nodes 0..n-1, the 2n SegTree above (set its combine
//   and ID; path queries need a COMMUTATIVE combine). Values on
//   NODES. Edge weights: store w(p, v) on the child v and skip the
//   top node: in queryPath use seg.query(pos[u] + 1, pos[v] + 1).
// TIME: O(n log n) build, path O(log^2 n), subtree O(log n).
struct HLD {
    vector<int> parent, depth, heavy, head, pos, sz;
    SegTree seg;
    HLD(const vector<vector<int>> &adj, const vector<ll> &value,
        int root = 0) : seg(vector<ll>(adj.size(), 0)) {
        int n = adj.size();
        parent.assign(n, -1); depth.assign(n, 0); heavy.assign(n, -1);
        head.assign(n, 0); pos.assign(n, 0); sz.assign(n, 1);
        vector<int> order = {root};                // BFS order
        for (int i = 0; i < (int)order.size(); i++) {
            int u = order[i];
            for (int v : adj[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    order.push_back(v);
                }
        }
        for (int i = n - 1; i > 0; i--) {          // children first
            int v = order[i], p = parent[v];
            sz[p] += sz[v];
            if (heavy[p] == -1 || sz[v] > sz[heavy[p]]) heavy[p] = v;
        }
        int cur = 0;                    // DFS order, heavy child first
        vector<int> st = {root};
        while (!st.empty()) {
            int v = st.back();
            st.pop_back();
            pos[v] = cur++;
            int p = parent[v];
            head[v] = (p != -1 && heavy[p] == v) ? head[p] : v;
            for (int c : adj[v])
                if (c != p && c != heavy[v]) st.push_back(c);
            if (heavy[v] != -1) st.push_back(heavy[v]);
        }
        for (int v = 0; v < n; v++) seg.update(pos[v], value[v]);
    }
    void setValue(int v, ll x) { seg.update(pos[v], x); }
    ll queryPath(int u, int v) {
        ll res = seg.ID;
        while (head[u] != head[v]) {
            if (depth[head[u]] < depth[head[v]]) swap(u, v);
            res = seg.combine(res, seg.query(pos[head[u]], pos[u] + 1));
            u = parent[head[u]];
        }
        if (depth[u] > depth[v]) swap(u, v);       // u = LCA
        return seg.combine(res, seg.query(pos[u], pos[v] + 1));
    }
    ll querySubtree(int v) { return seg.query(pos[v], pos[v] + sz[v]); }
};
