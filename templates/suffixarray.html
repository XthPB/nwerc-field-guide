Suffix array and LCP

Cap any suffix comparison by the requested substring lengths.

From the ICPC Code Templates section “Suffix Array + LCP”. Use the code with the types and conventions shown in the excerpt.

// Suffix array + LCP. USE: #distinct substrings, longest repeated
//   substring, longest common substring of two strings, k-th
//   substring in sorted order. TIME: O(n log^2 n) (n = 2e5: 0.1 s).
// sa[i] = start of the i-th smallest suffix
vector<int> suffixArray(const string &s) {
    int n = s.size();
    if (n == 0) return {};
    vector<int> sa(n), cls(n), tmp(n);  // cls = rank by first len
    for (int i = 0; i < n; i++) { sa[i] = i; cls[i] = s[i]; }
    for (int len = 1; ; len *= 2) {
        auto key = [&](int i) {
            return make_pair(cls[i], i + len < n ? cls[i + len] : -1);
        };
        sort(sa.begin(), sa.end(),
             [&](int a, int b) { return key(a) < key(b); });
        tmp[sa[0]] = 0;
        for (int i = 1; i < n; i++)
            tmp[sa[i]] = tmp[sa[i - 1]] + (key(sa[i - 1]) < key(sa[i]));
        cls = tmp;
        if (cls[sa[n - 1]] == n - 1) break;  // all ranks different
    }
    return sa;
}
// lcp[i] = common prefix of suffixes sa[i] and sa[i+1], O(n)
vector<int> lcpArray(const string &s, const vector<int> &sa) {
    int n = s.size();
    vector<int> pos(n), lcp(max(n - 1, 0));
    for (int i = 0; i < n; i++) pos[sa[i]] = i;
    int h = 0;
    for (int i = 0; i < n; i++) {     // suffixes from long to short
        if (pos[i] == n - 1) { h = 0; continue; }
        int j = sa[pos[i] + 1];
        while (i + h < n && j + h < n && s[i + h] == s[j + h]) h++;
        lcp[pos[i]] = h;
        if (h > 0) h--;
    }
    return lcp;
}
// #distinct substrings = n(n+1)/2 - sum(lcp)
// LCP(suffix a, suffix b) = min lcp[pos[a]..pos[b]-1] (sparse table)
// longest repeated substring = max lcp; two strings: s + '#' + t
