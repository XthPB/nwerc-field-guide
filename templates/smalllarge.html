Small-to-large subtree merging

Keep the larger underlying container and move the smaller one's entries. Replace the payload and define how duplicated keys combine before quoting the resulting running time.

From the ICPC Code Templates cheat sheet.

// Small-to-large merging. USE: per-subtree sets/maps (distinct
//   values, most frequent value, counts) for every node of a tree.
// Always pour the SMALLER container into the larger: every element
//   moves O(log n) times -> O(n log^2 n) with set/map.
// NEEDS: rootTree from above. Example: distinct colours per subtree.
vector<int> distinctColours(const vector<vector<int>> &adj,
                            const vector<int> &colour) {
    int n = adj.size();
    vector<int> parent, order;
    rootTree(0, adj, parent, order);
    vector<set<int>> bag(n);
    vector<int> answer(n);
    for (int i = n - 1; i >= 0; i--) {   // children before parents
        int v = order[i];
        bag[v].insert(colour[v]);        // children already merged
        answer[v] = bag[v].size();
        if (v == order[0]) break;
        int p = parent[v];
        if (bag[p].size() < bag[v].size()) swap(bag[p], bag[v]);
        for (int x : bag[v]) bag[p].insert(x);
        bag[v].clear();
    }
    return answer;
}
// same idea with map<value, count> for "most frequent" queries
