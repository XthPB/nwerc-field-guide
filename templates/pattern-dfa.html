Prefix automaton for a forbidden word

For a nonempty pattern of lowercase English letters, next[s][c] is the matching prefix length after appending c. State m means the forbidden word has just been completed. A DP that forbids it discards that transition. The table takes O(26m) time and space.

std::vector<std::array<int, 26>> prefix_automaton(const std::string& pattern) {
    const int m = static_cast<int>(pattern.size());
    std::vector<int> border(m);
    for (int i = 1; i < m; ++i) {
        int k = border[i - 1];
        while (k > 0 && pattern[i] != pattern[k]) k = border[k - 1];
        if (pattern[i] == pattern[k]) ++k;
        border[i] = k;
    }
    std::vector<std::array<int, 26>> next(m);
    for (int s = 0; s < m; ++s) {
        for (int c = 0; c < 26; ++c) {
            if (pattern[s] == char('a' + c)) next[s][c] = s + 1;
            else next[s][c] = s == 0 ? 0 : next[border[s - 1]][c];
        }
    }
    return next;
}
