Count bracket replacements or feasible intervals

Choose bracket_completions for the number of fillings modulo a positive integer. Choose possible_bracket_intervals for the number of intervals admitting at least one filling. Both functions take a string over opening brackets, closing brackets and question marks.

From the DP companion.

// Replace each '?' independently with '(' or ')'. Count balanced results.
inline int bracket_completions(const std::string& s, int mod) {
    assert(mod >= 1);
    int n = static_cast<int>(s.size());
    std::vector<i64> ways(n + 1), next(n + 1);
    ways[0] = 1 % mod;
    for (int i = 0; i < n; ++i) {
        std::fill(next.begin(), next.end(), 0);
        for (int open = 0; open <= i; ++open) {
            if (s[i] != ')')
                next[open + 1] = (next[open + 1] + ways[open]) % mod;
            if (s[i] != '(' && open > 0)
                next[open - 1] = (next[open - 1] + ways[open]) % mod;
        }
        ways.swap(next);
    }
    return static_cast<int>(ways[0]);
}

// Each nonempty interval counts once if at least one replacement works.
inline i64 possible_bracket_intervals(const std::string& s) {
    int n = static_cast<int>(s.size());
    i64 answer = 0;
    for (int left = 0; left < n; ++left) {
        int low = 0, high = 0;
        for (int right = left; right < n; ++right) {
            char c = s[right];
            high += (c == ')' ? -1 : 1);
            if (high < 0) break;
            if (c == '(') ++low;
            else low = (low == 0 ? 1 : low - 1);
            if (low == 0) ++answer;
        }
    }
    return answer;
}
