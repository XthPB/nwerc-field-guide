Adaptive Simpson integration

Handle singularities and split at known discontinuities before relying on a local error estimate.

From the ICPC Code Templates section “Adaptive Simpson Integration”. Use the code with the types and conventions shown in the excerpt.

// Adaptive Simpson: integral of f over [a, b] to absolute error ~eps.
// USE: areas / volumes with no formula (area of a union of circles:
//   integrate the covered length of each vertical line x).
// NEEDS: f smooth on [a, b]; split [a, b] at kinks (e.g. at every
//   circle's left/right x) and add the pieces.
double simpsonPart(const function<double(double)> &f, double a,
                   double b, double fa, double fm, double fb,
                   double whole, double eps, int depth) {
    double m = (a + b) / 2, flm = f((a + m) / 2), frm = f((m + b) / 2);
    double left = (m - a) / 6 * (fa + 4 * flm + fm);
    double right = (b - m) / 6 * (fm + 4 * frm + fb);
    if (depth <= 0 || fabs(left + right - whole) <= 15 * eps)
        return left + right + (left + right - whole) / 15;
    return simpsonPart(f, a, m, fa, flm, fm, left, eps / 2, depth - 1)
         + simpsonPart(f, m, b, fm, frm, fb, right, eps / 2, depth - 1);
}
double integrate(const function<double(double)> &f, double a,
                 double b, double eps = 1e-9) {
    double fa = f(a), fm = f((a + b) / 2), fb = f(b);
    double whole = (b - a) / 6 * (fa + 4 * fm + fb);
    return simpsonPart(f, a, b, fa, fm, fb, whole, eps, 40);
}
