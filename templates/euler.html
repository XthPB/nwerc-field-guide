Euler trails

Check connectivity on all non-isolated vertices and verify that the output uses every edge.

From the ICPC Code Templates section “Euler Path / Circuit”. Use the code with the types and conventions shown in the excerpt.

// Euler path / circuit (Hierholzer): a walk using EVERY EDGE once.
// USE: draw without lifting the pen, domino chains, De Bruijn
//   strings, route through every street once.
// EXISTS: DIRECTED: out = in at every node (circuit), or one start
//   with out = in + 1 and one end with in = out + 1 (path).
//   UNDIRECTED: 0 (circuit) or 2 (path) nodes of odd degree.
//   Plus: all edges in one connected piece (checked at the end).
// NEEDS: nodes 0..n-1, at least 1 edge. TIME: O(V + E), iterative.
// Returns the E + 1 nodes of the walk, or {} if there is none.
vector<int> eulerPath(int n, const vector<pair<int, int>> &edges,
                      bool directed) {
    int m = edges.size();
    vector<vector<pair<int, int>>> adj(n);    // {to, edge id}
    vector<int> deg(n, 0);                    // directed: out - in
    for (int i = 0; i < m; i++) {
        auto [u, v] = edges[i];
        adj[u].push_back({v, i});
        if (directed) { deg[u]++; deg[v]--; }
        else { adj[v].push_back({u, i}); deg[u]++; deg[v]++; }
    }
    int start = edges[0].first, plus = 0, minus = 0;
    for (int v = 0; v < n; v++) {
        if (directed) {
            if (deg[v] == 1) { plus++; start = v; }
            else if (deg[v] == -1) minus++;
            else if (deg[v] != 0) return {};
        } else if (deg[v] % 2 == 1) { plus++; start = v; }
    }
    if (directed ? (plus != minus || plus > 1) : plus > 2) return {};
    vector<bool> used(m, false);
    vector<int> ptr(n, 0), path, st = {start};
    while (!st.empty()) {
        int v = st.back();
        while (ptr[v] < (int)adj[v].size() &&
               used[adj[v][ptr[v]].second])
            ptr[v]++;                        // skip used edges
        if (ptr[v] == (int)adj[v].size()) {   // stuck: v is final
            path.push_back(v);
            st.pop_back();
        } else {
            auto [to, id] = adj[v][ptr[v]];
            used[id] = true;
            st.push_back(to);
        }
    }
    reverse(path.begin(), path.end());
    if ((int)path.size() != m + 1) return {};   // not connected
    return path;
}
