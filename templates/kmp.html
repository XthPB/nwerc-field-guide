Prefix-function matching

This guide adaptation accepts any characters in the pattern and text. An empty pattern matches every boundary, including both ends.

Adapted for this guide from the ICPC Code Templates section “KMP”. Use the code with the types and conventions shown in the excerpt.

std::vector<int> pi(const std::string& pattern) {
    std::vector<int> border(pattern.size());
    for (int i = 1; i < static_cast<int>(pattern.size()); ++i) {
        int j = border[i - 1];
        while (j && pattern[i] != pattern[j]) j = border[j - 1];
        if (pattern[i] == pattern[j]) ++j;
        border[i] = j;
    }
    return border;
}
std::vector<int> kmp(const std::string& pattern, const std::string& text) {
    std::vector<int> matches;
    if (pattern.empty()) {
        matches.resize(text.size() + 1);
        std::iota(matches.begin(), matches.end(), 0);
        return matches;
    }
    const auto border = pi(pattern);
    int j = 0;
    for (int i = 0; i < static_cast<int>(text.size()); ++i) {
        while (j && text[i] != pattern[j]) j = border[j - 1];
        if (text[i] == pattern[j]) ++j;
        if (j == static_cast<int>(pattern.size())) {
            matches.push_back(i - j + 1);
            j = border[j - 1];
        }
    }
    return matches;
}
