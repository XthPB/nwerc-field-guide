Topological order and DAG DP

Use a verified acyclic order. Process every outgoing transition after its source is final. Counting merges by addition and longest paths by maximum.

From the ICPC Code Templates cheat sheet.

// Topological sort (Kahn). USE: order tasks with dependencies, DP on
//   a DAG (longest path, #paths), detect a directed cycle.
// NEEDS: DIRECTED graph. TIME: O(V + E). Returns {} on a cycle
//   ANYWHERE: for paths from s, first keep only nodes reachable from s.
vector<int> topoSort(const vector<vector<int>> &adj) {
    int n = adj.size();
    vector<int> indeg(n, 0), order;
    for (int u = 0; u < n; u++)
        for (int v : adj[u]) indeg[v]++;
    queue<int> q;     // min-heap -> lexicographically smallest
    for (int u = 0; u < n; u++)
        if (indeg[u] == 0) q.push(u);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        order.push_back(u);
        for (int v : adj[u])
            if (--indeg[v] == 0) q.push(v);
    }
    if ((int)order.size() < n) return {};
    return order;
}
// DAG DP (longest path, #paths): for u in order, update u -> v
