Bellman–Ford and edge-count layers

For exactly t edges use a fresh layer after each pass. An in-place shortest-path relaxation can traverse several edges in a single pass and changes that meaning.

From the ICPC Code Templates cheat sheet.

// Bellman-Ford: NEGATIVE weights allowed, finds negative cycles.
// NEEDS: directed edges {u, v, w}, nodes 0..n-1. TIME: O(V E).
// dist = INF: unreachable, -INF: a negative cycle makes it -infinity
vector<ll> bellmanFord(int n, int src,
                       const vector<array<ll, 3>> &edges) {
    vector<ll> dist(n, INF);
    dist[src] = 0;
    for (int iter = 0; iter < n - 1; iter++)
        for (auto [u, v, w] : edges)
            if (dist[u] != INF && dist[u] + w < dist[v])
                dist[v] = dist[u] + w;
    for (int iter = 0; iter < n; iter++)    // spread -INF
        for (auto [u, v, w] : edges)
            if (dist[u] != INF &&
                (dist[u] == -INF || dist[u] + w < dist[v]))
                dist[v] = -INF;
    return dist;
}
// Floyd-Warshall: ALL PAIRS, V <= ~500, O(V^3), negative weights OK.
// Start d[i][i] = 0, d[i][j] = w (min over parallel edges) or INF.
// Afterwards d[i][i] < 0: i lies on a negative cycle.
void floydWarshall(vector<vector<ll>> &d) {
    int n = d.size();
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (d[i][k] < INF && d[k][j] < INF)
                    d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
}
