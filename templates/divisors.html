Factorization and divisors

Enumerate divisors to index a small state space. Exact gcd counts require subtracting proper multiples from all-multiples counts. Use overflow-safe multiplication when growing an LCM.

From the ICPC Code Templates cheat sheet.

// Factorisation by trial division, O(sqrt n) (n = 1e12: 1e6 steps).
// MANY numbers <= 1e7: loop p only over the primes of the Sieve.
vector<pair<ll, int>> factorize(ll n) {     // {prime, exponent}
    vector<pair<ll, int>> f;
    for (ll p = 2; p * p <= n; p++) {
        if (n % p != 0) continue;
        int e = 0;
        while (n % p == 0) { n /= p; e++; }
        f.push_back({p, e});
    }
    if (n > 1) f.push_back({n, 1});
    return f;
}
vector<ll> divisors(ll n) {        // sorted, O(sqrt n)
    vector<ll> small, big;
    for (ll d = 1; d * d <= n; d++) {
        if (n % d != 0) continue;
        small.push_back(d);
        if (d != n / d) big.push_back(n / d);
    }
    small.insert(small.end(), big.rbegin(), big.rend());
    return small;
}
ll phi(ll n) {      // Euler phi: #k in [1, n] with gcd(k, n) = 1
    ll result = n;
    for (ll p = 2; p * p <= n; p++) {
        if (n % p != 0) continue;
        while (n % p == 0) n /= p;
        result -= result / p;
    }
    if (n > 1) result -= result / n;
    return result;
}
